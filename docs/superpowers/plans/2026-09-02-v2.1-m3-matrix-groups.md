```ts
/**
 * 矩阵分组存储(PRD-v2.1 FR-6)
 * 职责:平台分组(名称 + 平台集合)的持久化,userData/auto-publish/matrix-groups.json
 * 设计要点:load/persist 注入照 schedule-store;同名覆盖 id 稳定;变更即落盘
 */
import { app } from 'electron';
import { join, dirname } from 'path';
import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'fs';
import { logger } from '../../utils/logger';
import type { MatrixGroup } from './types';

/** 分组名最大长度 */
export const GROUP_NAME_MAX_LEN = 30;

/** 全部合法平台(与 auto-publish adapters 一致) */
export const PLATFORMS = ['douyin', 'kuaishou', 'xiaohongshu', 'bilibili', 'shipinhao', 'spzx'] as const;

/**
 * 校验分组名称与平台集合
 * @returns 错误信息;合法返回 null
 */
export function validateGroup(name: string, platforms: string[]): string | null {
  if (!name || name.trim().length === 0) return '分组名称不能为空';
  if (!Array.isArray(platforms) || platforms.length === 0) return '至少勾选一个平台';
  for (const p of platforms) {
    if (!(PLATFORMS as readonly string[]).includes(p)) return `平台非法: ${p}`;
  }
  return null;
}

/** 名称单独校验(表单即时反馈用) */
export function validateGroupName(name: string): string | null {
  if (!name || name.trim().length === 0) return '分组名称不能为空';
  if (name.length > GROUP_NAME_MAX_LEN) return `分组名称过长(≤${GROUP_NAME_MAX_LEN} 字符)`;
  return null;
}

/** 依赖注入(单测注入内存实现) */
export interface MatrixGroupStoreDeps {
  load?: () => MatrixGroup[];
  persist?: (list: MatrixGroup[]) => void;
}

/** 默认持久化文件路径(userData/auto-publish/matrix-groups.json) */
function storeFile(): string {
  const dir = join(app.getPath('userData'), 'auto-publish');
  mkdirSync(dir, { recursive: true });
  return join(dir, 'matrix-groups.json');
}

function defaultLoad(): MatrixGroup[] {
  try {
    const fp = storeFile();
    if (!existsSync(fp)) return [];
    return JSON.parse(readFileSync(fp, 'utf8')) as MatrixGroup[];
  } catch (err) {
    logger.error(`[matrix-groups] 加载失败: ${err instanceof Error ? err.message : String(err)}`);
    return [];
  }
}

function defaultPersist(list: MatrixGroup[]): void {
  try {
    const fp = storeFile();
    mkdirSync(dirname(fp), { recursive: true });
    writeFileSync(fp, JSON.stringify(list, null, 2), 'utf8');
  } catch (err) {
    logger.error(`[matrix-groups] 持久化失败: ${err instanceof Error ? err.message : String(err)}`);
  }
}

/** 矩阵分组存储(按 name 索引,变更即落盘) */
export class MatrixGroupStore {
  private groups = new Map<string, MatrixGroup>();
  private readonly loadFn: () => MatrixGroup[];
  private readonly persistFn: (list: MatrixGroup[]) => void;
  private loaded = false;

  constructor(deps: MatrixGroupStoreDeps = {}) {
    this.loadFn = deps.load ?? defaultLoad;
    this.persistFn = deps.persist ?? defaultPersist;
  }

  private ensureLoaded(): void {
    if (this.loaded) return;
    for (const g of this.loadFn()) this.groups().set(g.name, g);
    this.loaded = true;
  }

  /** 内部 Map 访问(ensureLoaded 后使用) */
  private groups(): Map<string, MatrixGroup> {
    return this.groupsMap;
  }
  private groupsMap = new Map<string, MatrixGroup>();

  /** 落盘 */
  private flush(): void {
    this.persistFn(this.list());
  }

  /** 保存(同名覆盖,id 稳定) */
  save(name: string, platforms: string[]): MatrixGroup {
    const err = validateGroup(name, platforms);
    if (err) throw new Error(`matrix-groups:save 参数无效:${err}`);
    this.ensureLoaded();
    const existing = this.groups().get(name);
    const now = new Date().toISOString();
    const group: MatrixGroup = {
      id: existing?.id ?? `mg-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 8)}`,
      name,
      platforms: [...platforms],
      createdAt: existing?.createdAt ?? now,
      updatedAt: now,
    };
    this.groups().set(name, group);
    this.flush();
    return group;
  }

  /** 全量列表(按更新时间降序) */
  list(): MatrixGroup[] {
    this.ensureLoaded();
    return [...this.groups().values()].sort((a, b) => b.updatedAt.localeCompare(a.updatedAt));
  }

  /** 按名称删除,返回是否成功 */
  remove(name: string): boolean {
    this.ensureLoaded();
    const ok = this.groups().delete(name);
    if (ok) this.flush();
    return ok;
  }
}

/** 全局单例 */
export const matrixGroupStore = new MatrixGroupStore();
```

```ts
/**
 * 矩阵分组聚合(PRD-v2.1 FR-6)
 * 职责:把发布分析记录按「矩阵分组」聚合(近 N 天播放/互动/发布数)
 * 设计要点:
 *   - 纯函数:不依赖 electron,可 node:test 单测
 *   - 记录的最新一条采集即当前值(与 buildDashboard 口径一致)
 *   - 互动率 = (点赞+评论)/播放,仅在有播放的记录上计算,分组级为均值
 */
import type { AnalyticsRecord } from '../auto-publish/analytics-store';
import type { MatrixGroup } from './types';

/** 单分组聚合行 */
export interface GroupAggregate {
  groupId: string;
  name: string;
  /** 分组内各记录最新播放之和 */
  totalPlays: number;
  /** 分组内各记录最新点赞+评论之和 */
  totalEngagement: number;
  /** 近 N 天内首次采集的记录数 */
  published: number;
  /** 分组平均互动率 = Σ单条互动率 / 有效条数(播放>0);无有效条目为 undefined */
  engagementRate?: number;
}

/**
 * 按分组聚合近 days 天的记录
 * @param records 分析记录
 * @param groups 矩阵分组
 * @param days 时间窗口(天)
 * @param now 当前时间戳(毫秒)
 * @returns 每分组一行(与 groups 顺序一致)
 */
export function aggregateByGroup(
  records: AnalyticsRecord[],
  groups: MatrixGroup[],
  days: number,
  now: number = Date.now(),
): GroupAggregate[] {
  const cutoff = now - days * 86400000;
  return groups.map((g) => {
    let totalPlays = 0;
    let totalEngagement = 0;
    let published = 0;
    let rateSum = 0;
    let rateCount = 0;
    for (const r of records) {
      if (!g.platforms.includes(r.platform)) continue;
      const history = r.history ?? [];
      const latest = history.length > 0 ? history[history.length - 1] : undefined;
      const firstAt = history.length > 0 ? Date.parse(history[0].collectedAt) : NaN;
      if (Number.isFinite(firstAt) && firstAt >= cutoff) published++;
      if (latest) {
        totalPlays += latest.plays ?? 0;
        totalEngagement += (latest.likes ?? 0) + (latest.comments ?? 0);
        if ((latest.plays ?? 0) > 0) {
          rateSum += ((latest.likes ?? 0) + (latest.comments ?? 0)) / (latest.plays as number);
          rateCount++;
        }
      }
    }
    return {
      groupId: g.id,
      name: g.name,
      totalPlays,
      totalEngagement,
      published,
      engagementRate: rateCount > 0 ? rateSum / rateCount : undefined,
    };
  });
}
```

```ts
/** 字符 bigram 集合构造 */
function bigrams(s: string): Set<string> {
  const set = new Set<string>();
  for (let i = 0; i < s.length - 1; i++) set.add(s.slice(i, i + 2));
  return set;
}

/** 字符 bigram Dice 相似度:2×|A∩B| / (|A|+|B|);任一为空返回 0 */
export function bigramSimilarity(a: string, b: string): number {
  if (a.length < 2 || b.length < 2) return 0;
  const A = bigrams(a);
  const B = bigrams(b);
  let inter = 0;
  for (const g of A) if (B.has(g)) inter++;
  const denom = A.size + B.size;
  return denom === 0 ? 0 : (2 * inter) / denom;
}

/**
 * 内容-分组匹配建议:按「标题相似度 0.6 + 历史平均互动率 0.4」加权
 * @param title 新作品标题
 * @param records 历史发布分析记录
 * @param groups 矩阵分组
 * @param topN 返回条数(默认 3)
 */
export function suggestGroups(
  title: string,
  records: AnalyticsRecord[],
  groups: MatrixGroup[],
  topN = 3,
): { groupId: string; name: string; score: number }[] {
  const scored = groups.map((g) => {
    let simSum = 0;
    let simCount = 0;
    let rateSum = 0;
    let rateCount = 0;
    for (const r of records) {
      if (!g.platforms.includes(r.platform)) continue;
      simSum += bigramSimilarity(title, r.title);
      simCount++;
      const history = r.history ?? [];
      const latest = history.length > 0 ? history[history.length - 1] : undefined;
      if (latest && (latest.plays ?? 0) > 0) {
        rateSum += ((latest.likes ?? 0) + (latest.comments ?? 0)) / (latest.plays as number);
        rateCount++;
      }
    }
    const sim = simCount > 0 ? simSum / simCount : 0;
    const rate = rateCount > 0 ? rateSum / rateCount : 0;
    // 互动率天然很小(常见 <0.2),做尺度放大后参与加权,避免被标题相似度淹没
    const engagementNorm = Math.min(rateSum === 0 ? 0 : rateSum / rateCount / 0.2, 1);
    return {
      groupId: g.id,
      name: g.name,
      score: TITLE_SIM_WEIGHT * (simCount ? simSum / simCount : 0) + ENGAGEMENT_WEIGHT * engagementNorm,
    };
  });
  // 互动率归一化占位说明:engagementNorm = min(平均互动率 / 0.2, 1),0.2 为经验满分互动率
  return scored.sort((a, b) => b.score - a.score).slice(0, topN);
}

/** 无用引用清理:analytics 记录仅类型引用 */
export type { AnalyticsRecord };
```

实现要点(务必照此落盘):
1. `suggest.ts` 顶部 `import type { AnalyticsRecord } from '../auto-publish/analytics-store';` 与 `import type { MatrixGroup } from './types';`;
2. `bigramSimilarity` 用上方正确版本(带 `bigrams` 辅助函数),**删除占位版**;
3. `engagementNorm` 计算:`Math.min(avgRate / 0.2, 1)`,其中 `avgRate = rateCount > 0 ? rateSum / rateCount : 0`;常量 `export const ENGAGEMENT_RATE_NORM = 0.2;` 集中定义;
4. simCount=0(组无记录)时标题相似度记 0。

```ts
/**
 * 矩阵分组 IPC 注册(PRD-v2.1 FR-6/7)
 * 通道列表:
 *   matrix-groups:save   - 新增/更新分组
 *   matrix-groups:list   - 全量列表
 *   matrix-groups:delete - 按名称删除
 *   matrix:suggest       - 内容-分组匹配建议(可选 LLM 解释)
 */
import type { ipcMain } from 'electron';
import { safeHandle } from '../../../electron/ipc/index';
import {
  matrixGroupStore,
  validateGroup,
} from '../services/matrix/group-store';
import { aggregateByGroup } from '../services/matrix/matrix-dashboard';
import { suggestGroups } from '../services/matrix/suggest';
import type { MatrixGroup } from '../services/matrix/types';
import { analyticsStore } from '../services/auto-publish/analytics-store';
import { llmService } from '../services/llm';
import { logger } from '../utils/logger';

/**
 * 注册矩阵分组 IPC handlers
 * @param ipc ipcMain 实例
 */
export function register(ipc: typeof ipcMain): void {
  /**
   * 保存分组(同名覆盖)
   * payload: { name: string, platforms: string[] }
   * 返回: MatrixGroup
   */
  safeHandle(ipc, 'matrix-groups:save', async (_event, payload) => {
    const p = payload as { name?: string; platforms?: string[] } | undefined;
    if (!p?.name || !Array.isArray(p.platforms)) {
      throw new Error('matrix-groups:save 参数无效:缺少 name/platforms');
    }
    const err = validateGroup(p.name, p.platforms);
    if (err) throw new Error(`matrix-groups:save 参数无效:${err}`);
    return matrixGroupStore.save(p.name, JSON.parse(JSON.stringify(p.platforms)));
  });

  /** 全量列表 */
  safeHandle(ipc, 'matrix-groups:list', async () => matrixGroupStore.list());

  /**
   * 删除分组
   * payload: { name: string }
   */
  safeHandle(ipc, 'matrix-groups:delete', async (_event, payload) => {
    const p = payload as { name?: string } | undefined;
    if (!p?.name) throw new Error('matrix-groups:delete 缺少 name');
    return matrixGroupStore.remove(p.name);
  });

  /**
   * 内容-分组匹配建议
   * payload: { title: string, topN?: number, explain?: boolean }
   * 返回: { suggestions, explanation? }
   * 未配置 LLM 或调用失败时 explanation 为 undefined,不报错(纯函数建议兜底)
   */
  safeHandle(ipc, 'matrix:suggest', async (_event, payload) => {
    const p = payload as
      { title?: string; topN?: number; explain?: boolean; groups?: MatrixGroup[] }
      | undefined;
    if (!p?.title || typeof p.title !== 'string') {
      throw new Error('matrix:suggest 参数无效:缺少 title');
    }
    const groups = p.groups ?? matrixGroupStore.list();
    const records = analyticsStore.list();
    const suggestions = suggestGroups(p.title, records, groups, p.topN ?? 3);
    if (!p.explain) return { suggestions };

    // 可选 LLM 解释(容错降级:失败不报错)
    try {
      const prompt =
        `你是短视频矩阵运营助手。新作品标题:「${p.title}」。` +
        `候选分组及其近期数据:${JSON.stringify(suggestions)}。` +
        `请用不超过 80 字解释为何第一推荐最适合。只输出 JSON:{"reason": "..."}`;
      const resp = await llmService.chat({
        messages: [{ role: 'user', content: prompt }],
        temperature: 0.3,
        maxTokens: 200,
      });
      let reason = resp.content ?? '';
      reason = reason.replace(/```json|```/g, '').trim();
      try {
        const parsed = JSON.parse(reason) as { reason?: string };
        reason = parsed.reason ?? reason;
      } catch {
        /* 非 JSON 直接使用原文 */
      }
      return { suggestions, explanation: reason };
    } catch (err) {
      logger.warn(
        `[matrix] LLM 解释失败(降级为纯建议): ${err instanceof Error ? err.message : String(err)}`,
      );
      return { suggestions };
    }
  });
}
```
